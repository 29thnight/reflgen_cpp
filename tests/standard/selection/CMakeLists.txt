cmake_minimum_required(VERSION 3.25)
project(reflgen_backend_selection LANGUAGES CXX)
set(REFLGEN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(REFLGEN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(REFLGEN_BUILD_GENERATOR OFF CACHE BOOL "" FORCE)
add_subdirectory(../../.. reflgen)
reflgen_configure_language_modes()
add_executable(selection main.cpp)
target_link_libraries(selection PRIVATE reflgen::reflgen)
if(NOT TEST_STANDARD)
    set(TEST_STANDARD 20)
endif()
set_property(TARGET selection PROPERTY CXX_STANDARD ${TEST_STANDARD})
set_property(TARGET selection PROPERTY CXX_STANDARD_REQUIRED ON)
target_compile_features(selection PRIVATE cxx_std_${TEST_STANDARD})
target_compile_definitions(selection PRIVATE EXPECT_LANGUAGE=${TEST_STANDARD})
reflgen_select_backend(selection "${TEST_BACKEND}" selected)
if(TEST_REFLECTION_FLAG AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(selection PRIVATE -freflection)
endif()
if(TEST_EXPECTED AND NOT selected STREQUAL TEST_EXPECTED)
    message(FATAL_ERROR "Expected ${TEST_EXPECTED}, selected ${selected}")
endif()
if(selected STREQUAL "STD")
    target_compile_definitions(selection PRIVATE EXPECT_STANDARD=1)
else()
    target_compile_definitions(selection PRIVATE EXPECT_STANDARD=0)
endif()
message(STATUS "REFLGEN_SELECTION=${selected}")
enable_testing()
add_test(NAME selection_runtime COMMAND selection)
