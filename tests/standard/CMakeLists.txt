cmake_minimum_required(VERSION 3.28)
project(reflgen_standard_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(REFLGEN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(REFLGEN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(REFLGEN_BUILD_GENERATOR OFF CACHE BOOL "" FORCE)
option(REFLGEN_STANDARD_INSTALLED "Test an installed package instead of the source tree" OFF)
if(REFLGEN_STANDARD_INSTALLED)
    find_package(reflgen CONFIG REQUIRED)
else()
    add_subdirectory(../.. reflgen)
endif()
add_executable(standard_test runtime.cpp)
target_link_libraries(standard_test PRIVATE reflgen::reflgen)
reflgen_generate(standard_test BACKEND AUTO HEADERS types.h TYPES player MODULE standard_fixture)
get_target_property(selected standard_test REFLGEN_SELECTED_BACKEND)
if(NOT selected STREQUAL "STD")
    message(FATAL_ERROR "native fixture requires a compiler implementing the standard backend")
endif()
enable_testing()
add_test(NAME standard_runtime COMMAND standard_test)
add_executable(legacy_test legacy.cpp)
target_link_libraries(legacy_test PRIVATE reflgen::reflgen)
reflgen_select_backend(legacy_test LEGACY legacy_selected)
add_test(NAME legacy_opt_out COMMAND legacy_test)

add_library(standard_types STATIC)
target_sources(standard_types PUBLIC FILE_SET CXX_MODULES FILES types.ixx)
target_link_libraries(standard_types PUBLIC reflgen::reflgen)
reflgen_generate_ifc(native_registration PROVIDER standard_types IMPORT standard.types
    MODULE native_module TYPES native_module::actor BACKEND AUTO)
add_executable(module_test module_runtime.cpp)
target_link_libraries(module_test PRIVATE native_registration)
add_test(NAME standard_module COMMAND module_test)
reflgen_add_modules(standard_library BACKEND STD)
add_executable(module_json_test module_json_runtime.cpp)
target_link_libraries(module_json_test PRIVATE native_registration standard_library)
add_test(NAME standard_named_json COMMAND module_json_test)
